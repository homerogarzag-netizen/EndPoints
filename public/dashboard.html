Consulta los datos en tiempo real desde cualquier bot o script en 5 líneas de Python: 

import requests

# Resumen completo SPX, QQQ o SPY
data = requests.get("https://end-points.vercel.app/v1/market/summary?symbol=SPX").json()

print(f"SPX: {data['price']} | Rango: [{data['lower_bound']} - {data['upper_bound']}]")
print(f"Call Wall: {data['call_wall']} | Put Wall: {data['put_wall']} (Flip: {data['zero_gamma']})")
print(f"VIX: {data['vix']} ({data['trend_status']})")
